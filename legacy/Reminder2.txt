// 省略代碼：初始化、添加、複習、渲染、拖拽、夜間模式
// 新增滑動手勢
function addSwipeListeners(card, itemId) {
  let startX=0, currentX=0, swiping=false;
  card.addEventListener('touchstart', e=>{
    startX=e.touches[0].clientX; swiping=true; card.style.transition='none';
  });
  card.addEventListener('touchmove', e=>{
    if(!swiping) return;
    currentX=e.touches[0].clientX;
    const dx=currentX-startX;
    card.style.transform=`translateX(${dx}px)`;
  });
  card.addEventListener('touchend', e=>{
    swiping=false;
    const dx=currentX-startX;
    if(dx<-80){ // 左滑刪除
      card.style.transform='translateX(-100%)';
      setTimeout(()=>{items=items.filter(i=>i.id!==itemId);renderAll();localStorage.setItem("review_items",JSON.stringify(items));},300);
    } else if(dx>80){ // 右滑標記已複習
      reviewItem(itemId,5);
      card.style.transform='translateX(0)';
    } else {
      card.style.transform='translateX(0)';
    }
    card.style.transition='transform 0.3s ease';
  });
}
